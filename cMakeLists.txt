cmake_minimum_required(VERSION 3.20)
project(orderbook LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_compile_options(-Wall -Wextra -Wpedantic)

include(FetchContent)

# --- Catch2 for unit tests ---
FetchContent_Declare(
  Catch2
  GIT_REPOSITORY https://github.com/catchorg/Catch2.git
  GIT_TAG v3.5.4
)
FetchContent_MakeAvailable(Catch2)

# --- Google Benchmark ---
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(
  benchmark
  GIT_REPOSITORY https://github.com/google/benchmark.git
  GIT_TAG v1.9.4
)
FetchContent_MakeAvailable(benchmark)

option(ENABLE_ASAN "Enable Address and UB sanitizers" OFF)
if(ENABLE_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

# --- Core library ---
add_library(orderbook_lib
  src/order.cpp
  src/book.cpp
  src/fast_book.cpp
  src/engine.cpp
)

target_include_directories(orderbook_lib PUBLIC include)

# --- Tests ---
add_executable(orderbook_tests
  tests/test_order.cpp
  tests/test_engine.cpp
  tests/differential/test_snapshot.cpp
  tests/differential/test_differential.cpp
  tests/differential/test_market_data.cpp
)
target_include_directories(orderbook_tests PRIVATE tests/differential)
target_link_libraries(orderbook_tests PRIVATE orderbook_lib Catch2::Catch2WithMain)

# --- Differential Fuzzer CLI ---
add_executable(orderbook_fuzz
  tests/differential/fuzz_main.cpp
  tests/differential/test_snapshot.cpp
)
target_include_directories(orderbook_fuzz PRIVATE tests/differential)
target_link_libraries(orderbook_fuzz PRIVATE orderbook_lib)

# --- Benchmarks ---
add_executable(orderbook_bench
  benchmarks/bench_order.cpp
)
target_link_libraries(orderbook_bench PRIVATE orderbook_lib benchmark::benchmark)

# --- CLI ---
add_executable(orderbook_cli
  src/main.cpp
)
target_link_libraries(orderbook_cli PRIVATE orderbook_lib)

# ── Gateway (Task 4.2) ────────────────────────────────────────────────────────
# Requires Boost (headers: Asio + Beast) and nlohmann/json.
# Both are installed by MSYS2 pacman into C:/msys64/mingw64.
# We add that tree as a hint so find_package can locate the cmake configs.

set(MSYS2_MINGW64 "C:/msys64/mingw64" CACHE PATH "MSYS2 mingw64 sysroot")

# Boost 1.67+ ships cmake config files; Beast and Asio are header-only.
list(APPEND CMAKE_PREFIX_PATH "${MSYS2_MINGW64}")
find_package(Boost 1.80 QUIET COMPONENTS system)

# nlohmann_json ships its own cmake config
find_package(nlohmann_json 3.0 QUIET)

if(Boost_FOUND AND nlohmann_json_FOUND)
  message(STATUS "Building orderbook_gateway  (Boost ${Boost_VERSION},"
          " nlohmann_json ${nlohmann_json_VERSION})")

  add_executable(orderbook_gateway src/gateway.cpp)
  target_include_directories(orderbook_gateway PRIVATE
    "${MSYS2_MINGW64}/include"   # Boost + nlohmann headers
  )
  target_link_libraries(orderbook_gateway PRIVATE
    orderbook_lib
    nlohmann_json::nlohmann_json
  )
  # Boost.Asio / Beast are header-only; only system + ws2_32 needed on Windows
  if(WIN32)
    target_link_libraries(orderbook_gateway PRIVATE ws2_32 mswsock)
  endif()
  target_compile_definitions(orderbook_gateway PRIVATE
    BOOST_ASIO_HEADER_ONLY
    _WIN32_WINNT=0x0601       # Windows 7+ for Winsock2
    NOMINMAX
  )
  target_compile_options(orderbook_gateway PRIVATE -g0 -O2)
  target_link_options(orderbook_gateway PRIVATE -s)
else()
  if(NOT Boost_FOUND)
    message(WARNING
      "Boost not found — orderbook_gateway will NOT be built.\n"
      "Install via: pacman -S mingw-w64-x86_64-boost")
  endif()
  if(NOT nlohmann_json_FOUND)
    message(WARNING
      "nlohmann_json not found — orderbook_gateway will NOT be built.\n"
      "Install via: pacman -S mingw-w64-x86_64-nlohmann-json")
  endif()
endif()
